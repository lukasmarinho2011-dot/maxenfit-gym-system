# --- FUNÇÕES DE VALIDAÇÃO ---


def validar_nome(texto):
    return not any(char.isdigit() for char in texto)


def validar_cpf(texto):
    if any(char.isalpha() for char in texto):
        return False
    return len(texto) <= 14


def validar_numerico_e_simbolos(texto):
    return not any(char.isalpha() for char in texto)


# --- PERSISTÊNCIA DE DADOS (JSON) ---


def carregar_dados():
    if not os.path.exists(ARQUIVO_DADOS):
        return

    try:
        with open(ARQUIVO_DADOS, "r", encoding="utf-8") as arquivo:
            alunos = json.load(arquivo)

        for aluno in alunos:
            texto_item = montar_texto_item(aluno)
            listbox_alunos.insert(tk.END, texto_item)
    except (json.JSONDecodeError, IOError):
        messagebox.showwarning(
            "Aviso", "Não foi possível carregar os dados salvos anteriormente."
        )


def salvar_dados():
    alunos = []
    for item in listbox_alunos.get(0, tk.END):
        alunos.append(item)

    try:
        with open(ARQUIVO_DADOS, "w", encoding="utf-8") as arquivo:
            json.dump(alunos, arquivo, ensure_ascii=False, indent=2)
    except IOError:
        messagebox.showerror("Erro", "Não foi possível salvar os dados no arquivo.")


def montar_texto_item(aluno):
    if isinstance(aluno, str):
        return aluno
    return (
        f"{aluno['nome']} (CPF: {aluno['cpf']} | Nasc: {aluno['data_nasc']} | "
        f"{aluno['idade']} anos | Tel: {aluno['telefone']} | {aluno['peso']}kg | "
        f"{aluno['altura']}m | Obj: {aluno['objetivo']}) - Plano: {aluno['plano']}"
    )


def matricular():
    nome = entry_nome.get().strip()
    cpf = entry_cpf.get().strip()
    data_nasc = entry_data_nasc.get().strip()
    idade = entry_idade.get().strip()
    telefone = entry_telefone.get().strip()
    peso = entry_peso.get().strip()
    altura = entry_altura.get().strip()
    objetivo = entry_objetivo.get().strip()
    plano = entry_plano.get().strip()

    if (
        not nome
        or not cpf
        or not data_nasc
        or not idade
        or not telefone
        or not peso
        or not altura
        or not objetivo
        or not plano
    ):
        messagebox.showwarning("Aviso", "Por favor, preencha todos os campos!")
        return

    planos_validos = ["mensal", "trimestral", "anual"]
    if plano.lower() not in planos_validos:
        messagebox.showerror(
            "Erro no Plano",
            "O campo Plano aceita apenas uma das opções: Mensal, Trimestral ou Anual!",
        )
        return

    texto_item = f"{nome} (CPF: {cpf} | Nasc: {data_nasc} | {idade} anos | Tel: {telefone} | {peso}kg | {altura}m | Obj: {objetivo}) - Plano: {plano.capitalize()}"
    listbox_alunos.insert(tk.END, texto_item)

    salvar_dados()

    entry_nome.delete(0, tk.END)
    entry_cpf.delete(0, tk.END)
    entry_data_nasc.config(state="normal")
    entry_data_nasc.delete(0, tk.END)
    entry_data_nasc.config(state="readonly")
    entry_idade.delete(0, tk.END)
    entry_telefone.delete(0, tk.END)
    entry_peso.delete(0, tk.END)
    entry_altura.delete(0, tk.END)
    entry_objetivo.delete(0, tk.END)
    entry_plano.delete(0, tk.END)


def limpar_lista():
    resposta = messagebox.askyesno(
        "Confirmar", "Tem certeza que deseja apagar todos os alunos cadastrados?"
    )
    if resposta:
        listbox_alunos.delete(0, tk.END)
        salvar_dados()


# --- CALENDÁRIO PARA DATA DE NASCIMENTO ---


def abrir_calendario(event=None):
    valor_atual = entry_data_nasc.get().strip()
    if valor_atual and len(valor_atual) == 10:
        try:
            dia_s, mes_s, ano_s = valor_atual.split("/")
            calendario_estado["ano"] = int(ano_s)
            calendario_estado["mes"] = int(mes_s)
        except (ValueError, IndexError):
            calendario_estado["ano"] = date.today().year
            calendario_estado["mes"] = date.today().month
    else:
        calendario_estado["ano"] = date.today().year
        calendario_estado["mes"] = date.today().month

    popup = tk.Toplevel(root)
    popup.title("Selecionar Data de Nascimento")
    popup.configure(bg=COR_PRETO)
    popup.resizable(False, False)
    popup.transient(root)
    popup.grab_set()

    frame_topo = tk.Frame(popup, bg=COR_PRETO)
    frame_topo.pack(pady=(10, 5))

    btn_anterior = tk.Button(
        frame_topo,
        text="<",
        command=lambda: mudar_mes(-1),
        bg=COR_AZUL,
        fg=COR_TEXTO,
        relief="flat",
        width=3,
        cursor="hand2",
    )
    btn_anterior.grid(row=0, column=0, padx=5)

    label_mes_ano = tk.Label(
        frame_topo,
        text="",
        font=("Arial", 12, "bold"),
        bg=COR_PRETO,
        fg=COR_AZUL_CLARO,
        width=18,
    )
    label_mes_ano.grid(row=0, column=1)

    btn_proximo = tk.Button(
        frame_topo,
        text=">",
        command=lambda: mudar_mes(1),
        bg=COR_AZUL,
        fg=COR_TEXTO,
        relief="flat",
        width=3,
        cursor="hand2",
    )
    btn_proximo.grid(row=0, column=2, padx=5)

    frame_dias = tk.Frame(popup, bg=COR_PRETO)
    frame_dias.pack(padx=10, pady=(0, 10))

    def desenhar_calendario():
        for widget in frame_dias.winfo_children():
            widget.destroy()

        ano = calendario_estado["ano"]
        mes = calendario_estado["mes"]
        label_mes_ano.config(text=f"{MESES_PT[mes - 1]} {ano}")

        dias_semana = ["Seg", "Ter", "Qua", "Qui", "Sex", "Sáb", "Dom"]
        for col, dia_sem in enumerate(dias_semana):
            tk.Label(
                frame_dias,
                text=dia_sem,
                font=("Arial", 9, "bold"),
                bg=COR_PRETO,
                fg=COR_AZUL_CLARO,
                width=4,
            ).grid(row=0, column=col, padx=1, pady=1)

        hoje = date.today()
        semanas = calendar.monthcalendar(ano, mes)
        for linha, semana in enumerate(semanas, start=1):
            for col, dia in enumerate(semana):
                if dia == 0:
                    tk.Label(frame_dias, text="", bg=COR_PRETO, width=4).grid(
                        row=linha, column=col, padx=1, pady=1
                    )
                else:
                    data_dia = date(ano, mes, dia)
                    desabilitado = data_dia > hoje
                    btn_dia = tk.Button(
                        frame_dias,
                        text=str(dia),
                        width=4,
                        bg=COR_PRETO_CAMPO if not desabilitado else COR_PRETO,
                        fg=COR_TEXTO if not desabilitado else "#555555",
                        relief="flat",
                        state="disabled" if desabilitado else "normal",
                        cursor="hand2" if not desabilitado else "arrow",
                        command=lambda d=dia: selecionar_dia(d),
                    )
                    btn_dia.grid(row=linha, column=col, padx=1, pady=1)

    def mudar_mes(delta):
        mes = calendario_estado["mes"] + delta
        ano = calendario_estado["ano"]
        if mes > 12:
            mes = 1
            ano += 1
        elif mes < 1:
            mes = 12
            ano -= 1

        hoje = date.today()
        if (ano, mes) > (hoje.year, hoje.month):
            return

        calendario_estado["mes"] = mes
        calendario_estado["ano"] = ano
        desenhar_calendario()

    def selecionar_dia(dia):
        ano = calendario_estado["ano"]
        mes = calendario_estado["mes"]
        data_formatada = f"{dia:02d}/{mes:02d}/{ano}"
        entry_data_nasc.config(state="normal")
        entry_data_nasc.delete(0, tk.END)
        entry_data_nasc.insert(0, data_formatada)
        entry_data_nasc.config(state="readonly")
        popup.destroy()

    desenhar_calendario()

    popup.update_idletasks()
    x = root.winfo_x() + (root.winfo_width() // 2) - (popup.winfo_width() // 2)
    y = root.winfo_y() + (root.winfo_height() // 2) - (popup.winfo_height() // 2)
    popup.geometry(f"+{x}+{y}")


def desenhar_fundo(canvas, width, height, cor_circulos, cor_linhas):
    canvas.delete("fundo")

    for _ in range(15):
        x = random.randint(0, width)
        y = random.randint(0, height)
        raio = random.randint(10, 50)
        canvas.create_oval(
            x - raio,
            y - raio,
            x + raio,
            y + raio,
            outline=cor_circulos,
            width=1,
            tags="fundo",
        )

    for _ in range(10):
        x1 = random.randint(0, width)
        y1 = random.randint(0, height)
        x2 = random.randint(0, width)
        y2 = random.randint(0, height)
        canvas.create_line(
            x1, y1, x2, y2, fill=cor_linhas, width=1, dash=(4, 4), tags="fundo"
        )


def atualizar_fundo_abertura(event):
    desenhar_fundo(
        canvas_fundo_abertura, event.width, event.height, COR_PRETO_CAMPO, COR_AZUL
    )


def atualizar_fundo_principal(event):
    desenhar_fundo(
